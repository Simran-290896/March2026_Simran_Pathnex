## Ansible - Install Nginx

- name: Install Nginx
  hosts: all
  become: yes

  tasks:
    - name: Install nginx
      yum:
        name: nginx
	state: present

## Kubernetes - Deployment of 2 replicas

apiVersion: apps/v1
kind: Deployment
metadata:
  name: Pathnex Deployment
spec:
  replicas: 2
  selector:
    matchLabels:
      app: pathnex-app
  template:
    metadata:
      labels:
        app: pathnex-app
    spec:
      containers:
        - name: app
	  image: nginx
	    ports:
	      - container port: 80

## Jenkins Pipeline - Matrix Build

pipeline {
    agent any
    environment {
        INSTITUTE_NAME = "Pathnex"
    }
    matrix {
        axes {
	    axis {
	        name 'JAVA_VERSION'
	    Values '8', '11', '17'
	    } 
        }
        stages {
            stage('Build') {
	        steps {
	            sh echo "Building with Java $JAVA_VERSION"
	        }
	    }
        }
    }
}

## GitLab CI - Matrix Jobs

stages:
